前端 · 引入脚本（类似易盾 load.min.js）

<script src="{{$d.Endpoint}}/sdk/v1/load.js"></script>
<div id="captcha"></div>
<input type="hidden" name="botguard_token" id="token">
<script>
  initBotGuard({
    element: '#captcha',
    endpoint: '{{$d.Endpoint}}',
    appKey: '{{$a.AppKey}}',
    action: 'login',
    onVerify: function (err, ret) {
      if (!err) document.querySelector('#token').value = ret.token;
    }
  });
</script>

后端 · PHP SDK

// composer require botguard/php-sdk
$client = new \BotGuard\Client(
    '{{$d.Endpoint}}',
    getenv('BOTGUARD_SECRET_ID'),
    getenv('BOTGUARD_SECRET_KEY'),   // 只放服务端
    ['app_key' => '{{$a.AppKey}}']
);
$result = $client->verifyToken($_POST['botguard_token'] ?? null, 'login');
if (!$result->passed()) {
    // 拒绝本次请求，提示用户重新验证
}
