前端 · 引入脚本（类似易盾 load.min.js）

<script src="{{$d.Endpoint}}/sdk/v1/load.js"></script>
<div id="captcha"></div>
<input type="hidden" name="botguard_token" id="token">
<script>
  initBotGuard({
    element: '#captcha',
    endpoint: '{{$d.Endpoint}}',
    appKey: '{{$a.AppKey}}',
    action: 'login',
    onVerify: function (err, ret) {
      if (!err) document.querySelector('#token').value = ret.token;
    }
  });
</script>

前端 · 三种展示形式

// 在上面的 initBotGuard({...}) 里加 mode，标签写法同样用 mode="..."

// 触发式：鼠标移到组件上（触屏点一下）时，拼图浮在组件上方，不挤动页面
initBotGuard({ element: '#captcha', mode: 'float', endpoint: '{{$d.Endpoint}}', appKey: '{{$a.AppKey}}', action: 'login' });

// 嵌入式：组件可见时自动加载拼图，常驻在表单里（不写 mode 时点一下才展开）
initBotGuard({ element: '#captcha', mode: 'inline', endpoint: '{{$d.Endpoint}}', appKey: '{{$a.AppKey}}', action: 'login' });

// 弹出式：点提交按钮时弹出，通过后自动提交表单
initBotGuard({ mode: 'popup', bindTo: '#submit', endpoint: '{{$d.Endpoint}}', appKey: '{{$a.AppKey}}', action: 'login' });
